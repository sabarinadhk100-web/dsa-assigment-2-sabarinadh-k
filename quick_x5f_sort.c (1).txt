#include <stdio.h>

typedef struct {
    int id;
    int weight;
} Package;

long comparisons = 0;
int step=1;

void printPackages(Package arr[], int n){
    for(int i=0;i<n;i++) printf("P%d(%d) ", arr[i].id, arr[i].weight);
    printf("\n");
}

int partition(Package arr[], int low, int high){
    int pivot = arr[high].weight;
    int i = low-1;
    printf("  Step %d: Pivot = P%d(%d) [low=%d high=%d]\n", step++, arr[high].id, arr[high].weight, low, high);
    for(int j=low;j<high;j++){
        comparisons++;
        if(arr[j].weight <= pivot){
            i++;
            Package tmp=arr[i]; arr[i]=arr[j]; arr[j]=tmp;
        }
    }
    Package tmp=arr[i+1]; arr[i+1]=arr[high]; arr[high]=tmp;
    printf("  After partition: ");
    printPackages(arr, 8);
    return i+1;
}

void quickSort(Package arr[], int low, int high, int n){
    if(low<high){
        int pi = partition(arr, low, high);
        quickSort(arr, low, pi-1, n);
        quickSort(arr, pi+1, high, n);
    }
}

int main(){
    Package arr[8] = {{1,20},{2,15},{3,20},{4,10},{5,15},{6,20},{7,25},{8,10}};
    int n=8;
    printf("=== QUICK SORT - Package Weight Sorting ===\n");
    printf("Author: Sabari Nadh K\n");
    printf("Original order:\n");
    printPackages(arr,n);
    printf("\nIntermediate Steps:\n");
    quickSort(arr,0,n-1,n);
    printf("\nFinal Sorted (by weight):\n");
    printPackages(arr,n);
    printf("Total comparisons: %ld\n", comparisons);
    printf("\nStability Check (Quick Sort is NOT stable):\n");
    printf("Weight 10: order -> ");
    for(int i=0;i<n;i++) if(arr[i].weight==10) printf("P%d ", arr[i].id);
    printf("\nWeight 15: order -> ");
    for(int i=0;i<n;i++) if(arr[i].weight==15) printf("P%d ", arr[i].id);
    printf("\nWeight 20: order -> ");
    for(int i=0;i<n;i++) if(arr[i].weight==20) printf("P%d ", arr[i].id);
    printf("\n");
    return 0;
}
