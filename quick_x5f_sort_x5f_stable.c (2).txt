#include <stdio.h>

typedef struct {
    int id;
    int weight;
} Package;

long comparisons = 0;

void printPackages(Package arr[], int n){
    for(int i=0;i<n;i++) printf("P%d(%d) ", arr[i].id, arr[i].weight);
    printf("\n");
}

int partitionStable(Package arr[], int low, int high){
    Package pivot = arr[high];
    int i = low-1;
    for(int j=low;j<high;j++){
        comparisons++;
        if(arr[j].weight < pivot.weight || (arr[j].weight == pivot.weight && arr[j].id < pivot.id)){
            i++;
            Package tmp=arr[i]; arr[i]=arr[j]; arr[j]=tmp;
        }
    }
    Package tmp=arr[i+1]; arr[i+1]=arr[high]; arr[high]=tmp;
    return i+1;
}

void quickSortStable(Package arr[], int low, int high){
    if(low<high){
        int pi = partitionStable(arr, low, high);
        quickSortStable(arr, low, pi-1);
        quickSortStable(arr, pi+1, high);
    }
}

int main(){
    Package arr[8] = {{1,20},{2,15},{3,20},{4,10},{5,15},{6,20},{7,25},{8,10}};
    int n=8;
    printf("=== STABLE QUICK SORT (Modified) ===\n");
    printf("Author: Sabari Nadh K\n");
    printf("Original order:\n");
    printPackages(arr,n);
    printf("\nSorting with stability condition: if weight equal, smaller ID first\n");
    quickSortStable(arr,0,n-1);
    printf("\nFinal Sorted (Stable by ID):\n");
    printPackages(arr,n);
    printf("Total comparisons: %ld\n", comparisons);
    printf("\nVerification using Package IDs:\n");
    printf("Weight 10: Original P4,P8 -> Sorted: ");
    for(int i=0;i<n;i++) if(arr[i].weight==10) printf("P%d ", arr[i].id);
    printf(" [RETAINED]\n");
    printf("Weight 15: Original P2,P5 -> Sorted: ");
    for(int i=0;i<n;i++) if(arr[i].weight==15) printf("P%d ", arr[i].id);
    printf(" [RETAINED]\n");
    printf("Weight 20: Original P1,P3,P6 -> Sorted: ");
    for(int i=0;i<n;i++) if(arr[i].weight==20) printf("P%d ", arr[i].id);
    printf(" [RETAINED]\n");
    return 0;
}
